SYSTEM_ENV={'SYSTEMROOT','WINDIR','COMSPEC','PATH','PATHEXT','SYSTEMDRIVE','TEMP','TMP','LOCALAPPDATA','APPDATA',
    'USERPROFILE','PROGRAMDATA','PROGRAMFILES','PROGRAMFILES(X86)','NUMBER_OF_PROCESSORS','PROCESSOR_ARCHITECTURE',

CLEANUP_FAILURES={'dissolve_notify_failed','group_acl_delete_failed','queue_cleanup_failed','exchange_cleanup_failed'}
def now():return datetime.now(timezone.utc).isoformat()
def clean_environment():
    environment={name:value for name,value in os.environ.items() if name.upper() in SYSTEM_ENV}
    environment.update(PYTHONIOENCODING='utf-8',PYTHONUTF8='1',NO_PROXY='wt.ioa.pub,127.0.0.1,localhost')
    return environment
def run_child(argv,environment,gate,stem):
    with (gate/(stem+'.stdout.log')).open('wb') as stdout,(gate/(stem+'.stderr.log')).open('wb') as stderr:
        process=subprocess.Popen(argv,cwd=ROOT,env=environment,stdin=subprocess.DEVNULL,stdout=stdout,stderr=stderr)
        write(gate/(stem+'-started.json'),{'started_at':now(),'python_launcher_pid':process.pid,'argv':argv})
        return process.pid,process.wait()
def check_authorization(path,expected_sha,ready,plan):
    if path is None or expected_sha is None:raise ValueError('This NEW7 45 batch has no authorization; provide newly approved receipt and SHA')
    if sha(path)!=expected_sha:raise ValueError('Authorization bytes changed')
    value=read(path)
    expected={'schema_version':'incident-e3-execution-authorization/1.0','status':'AUTHORIZED',
        'scope_id':plan['scope_id'],'plan_sha256':sha(HERE/'plan.json'),'source_ready_sha256':sha(HERE/'source-ready.json'),
        'approved_cells':45,'allow_lab_high_repairs':True,'allow_separate_final_h00':True}
    if any(value.get(name)!=target for name,target in expected.items()):raise ValueError('New explicit authorization must bind this exact prepared scope, plan and source-ready bytes')
    return value
def matrix_checks(output,ready,plan):
    rows=read(output/'results.json')
    jsonl=[json.loads(line) for line in (output/'results.jsonl').read_text(encoding='utf-8').splitlines() if line.strip()]
    experiment=read(output/'experiment.json')
    fields=['scenario','method','repeat','workload_seed','precondition_pool_rebuilds','method_definition_id']
    actual=[{field:row[field] for field in fields} for row in rows]
    expected=[{field:row[field] for field in fields} for row in planned_cells()]
    checks={'exact_45_order_keys_and_metadata':actual==expected,
        'json_jsonl_equal':rows==jsonl,'frozen_source_manifest_exact':experiment.get('source_sha256')==ready['source_sha256'],
        'target_sha_exact':experiment.get('target_config_sha256')==sha(HERE/'target.json'),
        'transport_backend_exact':experiment.get('transport')=='wutong-amqps' and experiment.get('resource_backend')=='docker',
        'shared_model_alias_and_endpoint':experiment.get('model')=='Qwen3-1.7B-Q8_0' and experiment.get('model_base_url')=='http://127.0.0.1:8089/v1',
        'shared_sampling':experiment.get('sampling')=={'seed':42,'temperature':0,'top_p':1,'max_tokens':768},
        'shared_budgets':experiment.get('budgets')=={'tasks':40,'model_calls':40,'model_total_tokens':65536,'deadline_seconds':600},
        'M0_no_model_calls':all(row.get('model_enabled') is False and row.get('model_calls')==0 and row.get('tokens')==0 for row in rows if row['method']=='M0'),
        'M1_M2_enabled':all(row.get('model_enabled') is True for row in rows if row['method'] in {'M1','M2'})}
    confirmed=sum(row.get('status')=='completed' and row.get('continuous_recovery_confirmed') is True for row in rows)
    return {'checks':checks,'integrity_passed':all(checks.values()),'terminal_rows':len(rows),
        'confirmed_recovery_rows':confirmed,'all_45_recovered':len(rows)==45 and confirmed==45,
        'status_counts':dict(Counter(row.get('status') for row in rows))}
def close_checks(output,gate):
    events=[json.loads(line) for line in (output/'transport/events.jsonl').read_text(encoding='utf-8').splitlines() if line.strip()]
    created=[row for row in events if row.get('event')=='group_session_created']
    closed=[row for row in events if row.get('event')=='group_session_close_completed']
    warning_events=Counter(item for item in re.findall(r'event=([A-Za-z0-9_]+)',(gate/'matrix.stderr.log').read_text(encoding='utf-8',errors='replace')) if item in CLEANUP_FAILURES)
    matching=len(created)==len(closed)==1 and (created[0].get('session_id'),created[0].get('group_id'))==(closed[0].get('session_id'),closed[0].get('group_id'))
    return {'owned_close_event_pair':matching,'sdk_cleanup_warning_events':dict(warning_events),
        'client_closed_phase':any(row.get('event')=='client_close_phase' and row.get('phase')=='closed' for row in events),
        'passed':matching and not warning_events and any(row.get('event')=='client_close_phase' and row.get('phase')=='closed' for row in events),
        'limit':'Normal SDK close events and absence of captured cleanup warnings; no independent broker/ACL inventory query'}
